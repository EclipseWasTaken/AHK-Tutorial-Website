Classes & OOP

Proper class system with inheritance, properties, and static members.

Advanced 
class Animal {
    name := "Unknown"  ; instance property

    __New(name) {       ; constructor
        this.name := name
    }

    Speak() {
        return this.name . " says something"
    }

    ; Getter/setter property
    Label {
        get => "Animal: " . this.name
        set => this.name := value
    }

    static kingdom := "Animalia"  ; static member
}

class Dog extends Animal {
    breed := "Mixed"

    __New(name, breed) {
        super.__New(name)
        this.breed := breed
    }

    Speak() {
        return this.name . " barks!"
    }
}

fido := Dog("Fido", "Labrador")
MsgBox(fido.Speak())       ; "Fido barks!"
MsgBox(Animal.kingdom)    ; "Animalia"
MsgBox(fido is Animal)    ; true
