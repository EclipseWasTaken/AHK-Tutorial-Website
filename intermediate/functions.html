Functions

First-class functions, closures, fat-arrow syntax, and byref params.

; Basic function
Greet(name) {
    return "Hello, " . name . "!"
}
msg := Greet("Alice")

; Default parameter values
Power(base, exp := 2) {
    return base ** exp
}
Power(3)      ; = 9
Power(2, 8)  ; = 256

; Variadic — any number of args
Sum(args*) {
    total := 0
    For v in args
        total += v
    return total
}
Sum(1, 2, 3, 4)  ; = 10

; Fat-arrow (single expression)
Double := (n) => n * 2
Double(5)  ; = 10

; Closures
MakeAdder(n) {
    return (x) => x + n
}
Add5 := MakeAdder(5)
Add5(10)  ; = 15

; ByRef — modify caller's variable
Swap(&a, &b) {
    temp := a, a := b, b := temp
}
x := 1, y := 2
Swap(&x, &y)  ; x=2, y=1
