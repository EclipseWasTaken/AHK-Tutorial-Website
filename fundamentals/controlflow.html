Control Flow

Conditionals, loops, and flow control — all with clean C-style syntax.

If / Else

age := 25

if (age >= 18) {
    MsgBox("Adult")
} else if (age >= 13) {
    MsgBox("Teen")
} else {
    MsgBox("Child")
}

; Ternary
label := (age >= 18) ? "adult" : "minor"

; Logical operators
if (age >= 18 && age < 65)
    MsgBox("Working age")

Loops

; Loop N times — A_Index = 1..N
Loop 5 {
    ToolTip("Iteration: " . A_Index)
    Sleep(500)
}

; While loop
x := 0
While (x < 10) {
    x++
}

; For-in (arrays and maps)
fruits := ["Apple", "Banana", "Cherry"]
For index, fruit in fruits {
    MsgBox(index . ": " . fruit)
}

; Let's loop through all txt files on my Desktop
Loop Files, A_Desktop . "\*.txt" {
    MsgBox(A_LoopFileName)
}

; Here's how to use break and continue
Loop 10 {
    if (A_Index == 5)
        break
    if (Mod(A_Index, 2) == 0)
        continue
    MsgBox(A_Index)
}
