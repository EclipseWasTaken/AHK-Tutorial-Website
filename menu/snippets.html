Window snapping

; Snap active window to screen halves
#Left::  WinMove("A", , 0, 0, A_ScreenWidth//2, A_ScreenHeight)
#Right:: WinMove("A", , A_ScreenWidth//2, 0, A_ScreenWidth//2, A_ScreenHeight)
#Up::    WinMaximize("A")
#Down::  WinMinimize("A")

; Always-on-top toggle
^+t::
{
    id := WinExist("A")
    style := WinGetExStyle(id)
    if (style & 0x8)
        WinSetAlwaysOnTop(0, id)
    else
        WinSetAlwaysOnTop(1, id)
    ToolTip((style & 0x8) ? "Pinned OFF" : "Pinned ON")
    Sleep(1000)
    ToolTip()
}

📝 Text Expansion

#Requires AutoHotkey >= 2.0
#SingleInstance Force

; Basic expansions
::addr::"123 Main St, Springfield, IL 62701"
::eml::"yourname@example.com"
::ph::"(555) 867-5309"
::thx::"Thank you for reaching out. I'll get back to you shortly."

; Dynamic: insert today's date
::ddate::
{
    FormatTime(date, , "MMMM d, yyyy")
    Send(date)
}

; Multi-line signature
::sig::
{
    Send("Best regards,{Enter}Jane Smith{Enter}jane@company.com{Enter}+1 (555) 123-4567")
}

; Auto-fix typos (* = no end char needed)
:*:teh::"the"
:*:hte::"the"
:*:recieve::"receive"
:*:seperate::"separate"
:*:definately::"definitely"

; Paste clipboard as UPPERCASE
^+u::
{
    saved := A_Clipboard
    A_Clipboard := StrUpper(saved)
    Send("^v")
    A_Clipboard := saved
}

🖱️ Mouse & Clicks

; Click at specific coordinates
^F1::
{
    Click(500, 300)              ; Left click
    Click("Right", 500, 300)    ; Right click
    Click(500, 300, 2)           ; Double-click
}

; Show mouse coordinates (for finding click targets)
F1::
{
    MouseGetPos(&x, &y)
    ToolTip("X: " . x . "  Y: " . y)
    Sleep(2000)
    ToolTip()
}

; Wait for a pixel then click it
^F2::
{
    ToolTip("Waiting for green pixel...")
    While !PixelSearch(&px, &py, 0, 0, A_ScreenWidth, A_ScreenHeight, 0x00B050, 10)
        Sleep(100)
    ToolTip()
    Click(px, py)
}

; Keep-alive (jiggle mouse to prevent sleep)
keepAlive := false
^F9::
{
    global keepAlive
    keepAlive := !keepAlive
    if keepAlive
        SetTimer(Jiggle, 30000)
    else
        SetTimer(Jiggle, 0)
    TrayTip(keepAlive ? "Keep-alive ON" : "Keep-alive OFF")
}

Jiggle() {
    MouseGetPos(&x, &y)
    MouseMove(x+1, y+1)
    MouseMove(x, y)
}

🪟 Windows Management

; Focus app or launch it if not open
^F2::
{
    if WinExist("Notepad")
        WinActivate()
    else
        Run("notepad.exe")
}

; Window title criteria reference:
; "ahk_exe chrome.exe"    by process name
; "ahk_class Notepad"     by window class
; "ahk_pid 1234"          by process ID
; "ahk_id 0x12345"        by HWND

; List all open window titles
^F3::
{
    wins := WinGetList()
    list := ""
    For hwnd in wins {
        title := WinGetTitle(hwnd)
        if (title != "")
            list .= title . "`n"
    }
    MsgBox(list, "Open Windows")
}

; Center active window on screen
^+c::
{
    WinGetPos(&x, &y, &w, &h, "A")
    newX := (A_ScreenWidth - w) // 2
    newY := (A_ScreenHeight - h) // 2
    WinMove("A", , newX, newY)
}

; Launch app and type into it automatically
^F5::
{
    Run("notepad.exe")
    WinWait("Untitled - Notepad", , 10)
    WinActivate()
    Send("Hello from AHK!")
}
